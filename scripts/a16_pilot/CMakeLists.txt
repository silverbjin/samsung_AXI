cmake_minimum_required(VERSION 3.16)
project(SmartDishA16Pilot LANGUAGES CXX)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 8.15.0 EXACT REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(a16_pilot SHARED Pilot.cc)
target_compile_features(a16_pilot PRIVATE cxx_std_17)
target_link_libraries(a16_pilot PRIVATE gz-sim8::gz-sim8 gz-plugin2::gz-plugin2)

find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_executable(a16_native_contact NativeContact.cc)
target_compile_features(a16_native_contact PRIVATE cxx_std_17)
target_link_libraries(a16_native_contact PRIVATE gz-transport13::core gz-msgs10::gz-msgs10)
